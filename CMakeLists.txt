cmake_minimum_required(VERSION 3.31)
project(MeguvizCompute VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MGC_WITH_NCNN "Build pinned real ncnn provider and private SDK" ON)
option(MGC_VULKAN "Enable the ncnn-owned Vulkan domain" OFF)
option(MGC_BUILD_TESTS "Build independent host consumers" ON)
set(MGC_NCNN_SOURCE "" CACHE PATH "Verified ncnn source checkout")
set(MGC_GLSLANG_SOURCE "" CACHE PATH "Verified glslang checkout for Vulkan")
set(MGC_REQUIRED_PRIVATE_SDK_FINGERPRINT "" CACHE STRING "Exact expected private build identity")
set(MGC_NCNN_COMMIT "7e36d25cb153092ba87a60530455c7376fedf452")
set(MGC_GLSLANG_COMMIT "780b6a209132dc74e99458f62ebe5fa7e57c41cf")
if(MGC_WITH_NCNN)
  if(NOT EXISTS "${MGC_NCNN_SOURCE}/src/net.h")
    message(FATAL_ERROR "MGC_NCNN_SOURCE must point to the complete pinned ncnn checkout")
  endif()
  execute_process(COMMAND git -c "safe.directory=${MGC_NCNN_SOURCE}" -C "${MGC_NCNN_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE actual_ncnn OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual_ncnn STREQUAL MGC_NCNN_COMMIT)
    message(FATAL_ERROR "ncnn source commit mismatch: ${actual_ncnn}")
  endif()
  execute_process(COMMAND git -c "safe.directory=${MGC_NCNN_SOURCE}" -C "${MGC_NCNN_SOURCE}" diff --exit-code -- src cmake CMakeLists.txt RESULT_VARIABLE dirty_ncnn OUTPUT_QUIET)
  if(NOT dirty_ncnn EQUAL 0)
    message(FATAL_ERROR "Unregistered ncnn patch detected; update patch ledger first")
  endif()
  set(NCNN_VERSION 20261020 CACHE STRING "Pinned source release version" FORCE)
  set(NCNN_VULKAN ${MGC_VULKAN} CACHE BOOL "" FORCE)
  set(NCNN_INSTALL_SDK ON CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(NCNN_OPENMP OFF CACHE BOOL "" FORCE)
  set(NCNN_BUILD_WITH_STATIC_CRT ON CACHE BOOL "" FORCE)
  if(MGC_VULKAN)
    execute_process(COMMAND git -c "safe.directory=${MGC_GLSLANG_SOURCE}" -C "${MGC_GLSLANG_SOURCE}" rev-parse HEAD OUTPUT_VARIABLE actual_glslang OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT actual_glslang STREQUAL MGC_GLSLANG_COMMIT)
      message(FATAL_ERROR "glslang source commit mismatch")
    endif()
    set(ENABLE_GLSLANG_BINARIES OFF CACHE BOOL "" FORCE)
    set(ENABLE_HLSL OFF CACHE BOOL "" FORCE)
    set(ENABLE_OPT OFF CACHE BOOL "" FORCE)
    set(GLSLANG_TESTS OFF CACHE BOOL "" FORCE)
    set(GLSLANG_ENABLE_INSTALL ON CACHE BOOL "" FORCE)
    if(NOT EXISTS "${MGC_NCNN_SOURCE}/glslang/CMakeLists.txt")
      message(FATAL_ERROR "Initialize the pinned glslang submodule before a Vulkan build")
    endif()
    execute_process(COMMAND git -c "safe.directory=${MGC_NCNN_SOURCE}/glslang" -C "${MGC_NCNN_SOURCE}/glslang" rev-parse HEAD OUTPUT_VARIABLE nested_glslang OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT nested_glslang STREQUAL MGC_GLSLANG_COMMIT)
      message(FATAL_ERROR "ncnn glslang submodule identity mismatch")
    endif()
    set(NCNN_SYSTEM_GLSLANG OFF CACHE BOOL "" FORCE)
  endif()
  add_subdirectory("${MGC_NCNN_SOURCE}" ncnn)
  # ncnn's SIMD options apply to its own translation units. Public C ABI entries
  # always catch C++ exceptions before returning across the boundary.
endif()
find_package(Threads REQUIRED)
set(MGC_BUILD_OPTIONS "ncnn=${MGC_NCNN_COMMIT};glslang=${MGC_GLSLANG_COMMIT};patches=none;vulkan=${MGC_VULKAN};openmp=OFF;stdio=ON;string=ON;cxx=17;compiler=${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION};target=${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR};abi=${ANDROID_ABI};api=${ANDROID_PLATFORM};ndk=${CMAKE_ANDROID_NDK_VERSION};stl=${ANDROID_STL};ptr=${CMAKE_SIZEOF_VOID_P};config=${CMAKE_BUILD_TYPE};sdk=0.1.0")
file(SHA256 "${CMAKE_CXX_COMPILER}" compiler_file_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/private/meguviz/backend_ncnn_sdk.hpp" private_header_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/private/meguviz/compute_job.h" job_header_hash)
get_cmake_property(cache_keys CACHE_VARIABLES)
list(SORT cache_keys)
foreach(key IN LISTS cache_keys)
  if(key MATCHES "^NCNN_")
    string(APPEND MGC_BUILD_OPTIONS ";${key}=${${key}}")
  endif()
endforeach()
string(APPEND MGC_BUILD_OPTIONS ";with_ncnn=${MGC_WITH_NCNN};compiler_sha256=${compiler_file_hash};cflags=${CMAKE_C_FLAGS};cxxflags=${CMAKE_CXX_FLAGS};release_cxxflags=${CMAKE_CXX_FLAGS_RELEASE};private_headers=${private_header_hash}-${job_header_hash}")
if(MGC_WITH_NCNN)
  file(SHA256 "${CMAKE_CURRENT_BINARY_DIR}/ncnn/src/platform.h" platform_header_hash)
  string(APPEND MGC_BUILD_OPTIONS ";ncnn_platform_hash=${platform_header_hash}")
endif()
string(SHA256 MGC_PRIVATE_FINGERPRINT "${MGC_BUILD_OPTIONS}")
if(NOT MGC_WITH_NCNN)
  set(MGC_PRIVATE_FINGERPRINT "")
endif()
if(MGC_REQUIRED_PRIVATE_SDK_FINGERPRINT AND NOT MGC_REQUIRED_PRIVATE_SDK_FINGERPRINT STREQUAL MGC_PRIVATE_FINGERPRINT)
  message(FATAL_ERROR "private SDK fingerprint mismatch")
endif()
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/include/meguviz/compute.h" public_header_hash)
file(GLOB_RECURSE library_inputs CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/include/*" "${CMAKE_CURRENT_SOURCE_DIR}/private/*" "${CMAKE_CURRENT_SOURCE_DIR}/src/*")
list(SORT library_inputs)
set(library_source_hashes "")
foreach(input IN LISTS library_inputs)
  file(SHA256 "${input}" digest)
  string(APPEND library_source_hashes "${digest};")
endforeach()
string(SHA256 MGC_BUILD_FINGERPRINT "${MGC_PRIVATE_FINGERPRINT};${public_header_hash};${library_source_hashes}")
configure_file(cmake/build-info.json.in build-info.json @ONLY)
add_library(meguviz_compute_core STATIC src/core/runtime.cpp)
set_target_properties(meguviz_compute_core PROPERTIES EXPORT_NAME core CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
add_library(MeguvizCompute::core ALIAS meguviz_compute_core)
target_include_directories(meguviz_compute_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE private)
target_compile_definitions(meguviz_compute_core PRIVATE MGC_HAVE_NCNN=$<BOOL:${MGC_WITH_NCNN}> MGC_BUILD_FP="${MGC_BUILD_FINGERPRINT}" MGC_PRIVATE_FP="${MGC_PRIVATE_FINGERPRINT}")
target_link_libraries(meguviz_compute_core PUBLIC Threads::Threads)
if(MGC_WITH_NCNN)
  target_link_libraries(meguviz_compute_core PRIVATE ncnn)
endif()
if(MSVC)
  target_compile_options(meguviz_compute_core PRIVATE /EHsc /W4)
  set_property(TARGET meguviz_compute_core PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(meguviz_compute_core PRIVATE -Wall -Wextra -Wpedantic)
  set_source_files_properties(src/core/runtime.cpp PROPERTIES COMPILE_OPTIONS "-fexceptions;-frtti")
endif()
add_library(meguviz_compute_kernels INTERFACE)
set_target_properties(meguviz_compute_kernels PROPERTIES EXPORT_NAME kernels)
target_link_libraries(meguviz_compute_kernels INTERFACE meguviz_compute_core)
add_library(MeguvizCompute::kernels ALIAS meguviz_compute_kernels)
if(MGC_WITH_NCNN)
  add_library(meguviz_compute_ncnn_sdk INTERFACE)
  set_target_properties(meguviz_compute_ncnn_sdk PROPERTIES EXPORT_NAME backend_ncnn_sdk)
  target_include_directories(meguviz_compute_ncnn_sdk INTERFACE $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/private> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/meguviz-private>)
  target_link_libraries(meguviz_compute_ncnn_sdk INTERFACE meguviz_compute_core ncnn)
  set_property(TARGET ncnn PROPERTY MGC_COMPUTE_PRIVATE_SDK "${MGC_PRIVATE_FINGERPRINT}")
  set_property(TARGET meguviz_compute_ncnn_sdk PROPERTY INTERFACE_MGC_PRIVATE_SDK_FINGERPRINT "${MGC_PRIVATE_FINGERPRINT}")
  set_property(TARGET meguviz_compute_ncnn_sdk APPEND PROPERTY COMPATIBLE_INTERFACE_STRING MGC_PRIVATE_SDK_FINGERPRINT)
  add_library(MeguvizCompute::backend_ncnn_sdk ALIAS meguviz_compute_ncnn_sdk)
  install(TARGETS meguviz_compute_ncnn_sdk EXPORT MeguvizComputeTargets)
  install(DIRECTORY private/meguviz DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/meguviz-private)
endif()
install(TARGETS meguviz_compute_core meguviz_compute_kernels EXPORT MeguvizComputeTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MeguvizComputeTargets NAMESPACE MeguvizCompute:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizCompute)
configure_package_config_file(cmake/MeguvizComputeConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/MeguvizComputeConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizCompute)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MeguvizComputeConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MeguvizComputeConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/MeguvizComputeConfigVersion.cmake" "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MeguvizCompute)
install(FILES NOTICE.md LICENSE third_party/SBOM.json third_party/locks/ncnn.lock.json DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizCompute)
install(DIRECTORY schemas/shaderpack DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizCompute)
if(MGC_WITH_NCNN)
  install(FILES "${MGC_NCNN_SOURCE}/LICENSE.txt" DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizCompute/licenses RENAME ncnn-LICENSE.txt)
  if(MGC_VULKAN)
    install(FILES "${MGC_NCNN_SOURCE}/glslang/LICENSE.txt" DESTINATION ${CMAKE_INSTALL_DATADIR}/MeguvizCompute/licenses RENAME glslang-LICENSE.txt)
  endif()
endif()

  if(MGC_WITH_NCNN)

  endif()
  if(MSVC)
    foreach(t mgc_c_consumer mgc_contract_tests mgc_tensor_statistics mgc_headless mgc_ncnn_c_graph mgc_core_conformance)
      if(TARGET ${t})
        set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
      endif()

  endif()
